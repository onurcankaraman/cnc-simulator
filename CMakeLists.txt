cmake_minimum_required(VERSION 3.16)
project(cnc_simulator VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
find_package(Threads REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(open62541 CONFIG REQUIRED)
find_library(SNAP7 snap7 REQUIRED)
find_library(MOSQUITTO mosquitto REQUIRED)
add_library(cnc_core src/config.cpp src/core/engine.cpp src/transport/snapshot.cpp src/adapters/sqlite_store.cpp)
target_include_directories(cnc_core PUBLIC src /usr/local/include)
target_link_libraries(cnc_core PUBLIC yaml-cpp nlohmann_json::nlohmann_json SQLite::SQLite3 Threads::Threads)
target_compile_options(cnc_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cnc-simulator src/main.cpp src/adapters/protocols.cpp src/adapters/http_server.cpp)
target_link_libraries(cnc-simulator PRIVATE cnc_core open62541::open62541 ${SNAP7} ${MOSQUITTO})
add_executable(unit-tests tests/unit/tests.cpp)
target_link_libraries(unit-tests PRIVATE cnc_core)
enable_testing()
add_test(NAME deterministic COMMAND unit-tests WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
install(TARGETS cnc-simulator RUNTIME DESTINATION bin)
add_executable(stream-memory tests/unit/stream_memory.cpp)
target_link_libraries(stream-memory PRIVATE cnc_core)
add_test(NAME bounded_history COMMAND stream-memory WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
